Stack Implementation

Full JavaScript implementation with all stack operations.

Stack.js 
class Stack {
    constructor() {
        this.items = [];
        this.top = -1;
        this.maxSize = 10;
    }

    // Push operation
    push(element) {
        if (this.isFull()) {
            throw new Error("Stack Overflow");
        }
        this.top++;
        this.items[this.top] = element;
        return element;
    }

    // Pop operation
    pop() {
        if (this.isEmpty()) {
            throw new Error("Stack Underflow");
        }
        const element = this.items[this.top];
        this.top--;
        return element;
    }

    // Peek operation
    peek() {
        if (this.isEmpty()) {
            throw new Error("Stack is empty");
        }
        return this.items[this.top];
    }

    isEmpty() { return this.top === -1; }
    isFull() { return this.top === this.maxSize - 1; }
    size() { return this.top + 1; }
}
